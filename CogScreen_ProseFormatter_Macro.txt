============================================================
 FORMAT AS PROSE MACRO — Setup Instructions
============================================================

This macro reads your filled-in CogScreen tracking note and
converts it into flowing prose in your boss's preferred style.
Everything stays local in Word — no data leaves your machine.

This version (v3) reads fields by their stable tag name instead
of matching the title text, so renaming a title in the template
won't break the macro. It also uses Word's built-in placeholder
API to tell whether a field is empty, and substitutes the YP's
first name throughout if you've filled in that field at the top.

SETUP (replaces any earlier FormatAsProse you installed):
------------------------------------------------------------
1. Alt + F11 to open VBA editor
2. Open your Normal module
3. Delete any previous FormatAsProse, GetField, SafeGet, and
   AddPeriod subs/functions if they exist
4. Paste everything from the VBA CODE section below
5. Ctrl + S to save, Alt + Q to close

TO RUN:
------------------------------------------------------------
1. Fill in your tracking note template as usual (including the
   YP first name field at the top if you want the name used)
2. Run StripNAFields first (to remove N/A fields)
3. Then run FormatAsProse (Fn + Alt + F8 > FormatAsProse)
4. A new document will open with the prose version
5. Copy-paste from there into your system

============================================================
 VBA CODE — Copy everything below this line
============================================================

Sub FormatAsProse()
    '
    ' FormatAsProse v3
    ' Reads a filled CogScreen template by tag and outputs
    ' flowing prose in a new document. Substitutes the YP's
    ' first name (if provided) for "YP" throughout.
    '
    Dim doc As Document
    Set doc = ActiveDocument

    ' ── Read every known field by tag ──
    Dim ypName As String: ypName = GetField("yp_first_name")
    Dim subject As String
    If Len(ypName) > 0 Then
        subject = ypName
    Else
        subject = "YP"
    End If

    Dim sessionDate As String: sessionDate = GetField("session_date")
    Dim sessionTime As String: sessionTime = CleanTime(GetField("session_time"))
    Dim location As String:    location    = GetField("location")
    Dim appearance As String:  appearance  = GetField("appearance")
    Dim eyeContact As String:  eyeContact  = GetField("eye_contact")
    Dim affect As String:      affect      = GetField("affect")
    Dim rapport As String:     rapport     = GetField("rapport")
    Dim priorEvents As String: priorEvents = GetField("prior_events")

    Dim consentQ As String:       consentQ       = GetField("consent_questions")
    Dim consentSigned As String:  consentSigned  = GetField("consent_signed")
    Dim aceIssues As String:      aceIssues      = GetField("ace_issues")
    Dim aceFirstThree As String:  aceFirstThree  = GetField("ace_first_three")

    Dim neurocogReaction As String: neurocogReaction = GetField("neurocog_reaction")
    Dim neurocogMood As String:     neurocogMood     = GetField("neurocog_mood")
    Dim accommodations As String:   accommodations   = GetField("accommodations")
    Dim differences As String:      differences      = GetField("differences")

    Dim engagement As String: engagement = GetField("engagement")
    Dim clinical As String:   clinical   = GetField("clinical_impressions")
    Dim protocol As String:   protocol   = GetField("protocol")
    Dim risk As String:       risk       = GetField("risk")
    Dim sessionEnd As String: sessionEnd = GetField("session_end")
    Dim followUp As String:   followUp   = GetField("follow_up_support")

    ' ── Build the prose ──
    Dim prose As String

    ' Opening — substitute name for the first "YP" only
    prose = subject & " consented to participate in the CogScreen (HREC #2022.275) " & _
            "research study. Their participation was supported by their case manager " & _
            "who was consulted prior to any research contact. A summary of their " & _
            "research results (a cognitive summary) will be provided to the case " & _
            "manager to share with YP as per the protocol and consent procedure. "

    ' Session details — date/time/location
    If Len(sessionDate) > 0 Then
        prose = prose & "The session was conducted on " & sessionDate
        If Len(sessionTime) > 0 Then prose = prose & " at " & sessionTime
        If Len(location) > 0 Then prose = prose & " at " & location
        prose = prose & ". "
    ElseIf Len(location) > 0 Then
        prose = prose & "The session was conducted at " & location & ". "
    End If

    ' Appearance — prefix with subject, lowercase first letter
    If Len(appearance) > 0 Then
        prose = prose & PrefixSubject(subject, appearance)
        prose = EnsureTerminated(prose)
    End If

    ' Eye contact
    If Len(eyeContact) > 0 Then
        prose = prose & "Eye contact was " & LCase(eyeContact) & ". "
    End If

    ' Affect — use as typed (sentence fragment describing affect)
    If Len(affect) > 0 Then
        prose = prose & affect
        prose = EnsureTerminated(prose)
    End If

    ' Rapport
    If Len(rapport) > 0 Then
        prose = prose & "Rapport was " & LCase(rapport) & ". "
    End If

    ' Prior events — skip if "no", "nil", "nothing of note"
    If Len(priorEvents) > 0 Then
        Dim peLower As String: peLower = LCase(priorEvents)
        If peLower <> "nothing of note" And peLower <> "no" And peLower <> "nil" And _
           peLower <> "none" Then
            prose = prose & "Prior to the session, " & LCaseFirst(priorEvents)
            prose = EnsureTerminated(prose)
        End If
    End If

    ' Consent (Day One)
    If Len(consentQ) > 0 Then
        prose = prose & "Regarding the consent process, " & LCaseFirst(consentQ)
        prose = EnsureTerminated(prose)
    End If
    If Len(consentSigned) > 0 Then
        prose = prose & "The consent form was " & LCase(consentSigned) & ". "
    End If

    ' ACE (Day One)
    If Len(aceIssues) > 0 Then
        prose = prose & "Regarding ACE, " & LCaseFirst(aceIssues)
        prose = EnsureTerminated(prose)
    End If
    If Len(aceFirstThree) > 0 Then
        prose = prose & "For the first three ACE questions, " & LCase(aceFirstThree) & ". "
    End If

    ' Neurocog (Day Two / Test Retest)
    If Len(neurocogReaction) > 0 Then
        prose = prose & "During the neurocognitive assessment, " & LCaseFirst(neurocogReaction)
        prose = EnsureTerminated(prose)
    End If
    If Len(neurocogMood) > 0 Then
        prose = prose & subject & ChrW(8217) & "s mood during neurocog was " & _
                LCase(neurocogMood) & ". "
    End If
    If Len(accommodations) > 0 Then
        prose = prose & "Accommodations included: " & LCaseFirst(accommodations)
        prose = EnsureTerminated(prose)
    End If

    ' Test Retest
    If Len(differences) > 0 Then
        prose = prose & "Compared to the previous session, " & LCaseFirst(differences)
        prose = EnsureTerminated(prose)
    End If

    ' Engagement — prefix with subject
    If Len(engagement) > 0 Then
        prose = prose & PrefixSubject(subject, engagement)
        prose = EnsureTerminated(prose)
    End If

    ' Clinical impressions
    If Len(clinical) > 0 Then
        prose = prose & "Clinically, " & LCaseFirst(clinical)
        prose = EnsureTerminated(prose)
    End If

    ' Protocol
    If Len(protocol) > 0 Then
        If Left(LCase(protocol), 3) = "yes" Then
            prose = prose & "The session went according to protocol. "
        Else
            prose = prose & protocol
            prose = EnsureTerminated(prose)
        End If
    End If

    ' Risk
    If Len(risk) > 0 Then
        If InStr(1, LCase(risk), "nil", vbTextCompare) > 0 Or _
           InStr(1, LCase(risk), "no risk", vbTextCompare) > 0 Then
            prose = prose & "No risks were identified. "
        Else
            prose = prose & risk
            prose = EnsureTerminated(prose)
        End If
    End If

    ' Session end
    If Len(sessionEnd) > 0 Then
        prose = prose & "The session ended " & LCase(sessionEnd) & ". "
    End If

    ' Follow-up support
    If Len(followUp) > 0 Then
        prose = prose & "Follow-up: " & LCaseFirst(followUp)
        prose = EnsureTerminated(prose)
    End If

    ' ── Clean up double punctuation/spaces ──
    prose = Replace(prose, "..", ".")
    prose = Replace(prose, ". .", ".")
    prose = Replace(prose, "  ", " ")
    prose = Trim(prose)

    ' ── Output to a new document ──
    Dim newDoc As Document
    Set newDoc = Documents.Add
    newDoc.Content.Text = prose
    newDoc.Content.Font.Name = "Arial"
    newDoc.Content.Font.Size = 11

    MsgBox "Prose version created in new document. Review and copy-paste.", _
           vbInformation, "CogScreen Prose Formatter"
End Sub

' ── Read a content control's text by tag. ──
' Returns "" if missing, showing placeholder, or containing N/A.
Function GetField(tag As String) As String
    Dim ccs As ContentControls
    Set ccs = ActiveDocument.SelectContentControlsByTag(tag)
    If ccs.Count = 0 Then
        GetField = ""
        Exit Function
    End If

    Dim cc As ContentControl
    Set cc = ccs(1)

    ' Unfilled controls still show their placeholder text
    If cc.ShowingPlaceholderText Then
        GetField = ""
        Exit Function
    End If

    Dim txt As String
    txt = Trim(cc.Range.Text)

    ' Safety net: if the user didn't run StripNAFields, still skip N/A
    If InStr(1, txt, "N/A", vbTextCompare) > 0 Then
        GetField = ""
        Exit Function
    End If

    GetField = txt
End Function

' ── Strip a "Time:" label if the user left it in the time field ──
Function CleanTime(raw As String) As String
    Dim t As String
    t = raw
    If InStr(1, t, "Time:", vbTextCompare) > 0 Then
        t = Replace(t, "Time:", "", , , vbTextCompare)
    End If
    CleanTime = Trim(t)
End Function

' ── Ensure the running prose ends with ". " ──
Function EnsureTerminated(s As String) As String
    Dim trimmed As String
    trimmed = RTrim(s)
    Dim last As String
    last = Right(trimmed, 1)
    If last = "." Or last = "!" Or last = "?" Then
        EnsureTerminated = trimmed & " "
    Else
        EnsureTerminated = trimmed & ". "
    End If
End Function

' ── Lowercase first letter only; preserve rest ──
Function LCaseFirst(s As String) As String
    If Len(s) = 0 Then
        LCaseFirst = s
    ElseIf Len(s) = 1 Then
        LCaseFirst = LCase(s)
    Else
        LCaseFirst = LCase(Left(s, 1)) & Mid(s, 2)
    End If
End Function

' ── Prepend the subject (YP's name or "YP"), stripping any leading
'    "YP " or "<name> " the user may have already typed. ──
Function PrefixSubject(subject As String, text As String) As String
    Dim stripped As String
    stripped = text

    ' Remove a leading "YP " regardless of case
    If Len(stripped) > 3 And UCase(Left(stripped, 3)) = "YP " Then
        stripped = Mid(stripped, 4)
    End If

    ' Remove a leading "<subject> " (case insensitive) to avoid doubling
    If subject <> "YP" And Len(subject) > 0 Then
        Dim prefix As String
        prefix = subject & " "
        If Len(stripped) > Len(prefix) Then
            If LCase(Left(stripped, Len(prefix))) = LCase(prefix) Then
                stripped = Mid(stripped, Len(prefix) + 1)
            End If
        End If
    End If

    PrefixSubject = subject & " " & LCaseFirst(stripped)
End Function
